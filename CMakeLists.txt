cmake_minimum_required(VERSION 3.21)

project(keyboard_changer VERSION 1.1.0 LANGUAGES CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Widgets only: no QML / Quick runtime -> small memory footprint for a tray app.
# Qt 6.4+ is the main target. Qt 5.15 is supported for legacy builds:
# Windows 7 / 8 / 8.1 and 32-bit (x86) systems that Qt 6 no longer runs on.
set(KC_QT_COMPONENTS Widgets Network LinguistTools)
if(UNIX AND NOT APPLE)
    list(APPEND KC_QT_COMPONENTS DBus)
endif()
# Qt 6.4+ if present, otherwise Qt 5.15 (e.g. Ubuntu 22.04 ships only Qt 6.2).
option(KC_FORCE_QT5 "Build with Qt 5.15 even if Qt 6 is available" OFF)
if(NOT KC_FORCE_QT5)
    find_package(Qt6 6.4 QUIET COMPONENTS Core)
endif()
if(Qt6_FOUND)
    set(QT_VERSION_MAJOR 6)
    set(QT_VERSION ${Qt6_VERSION})
else()
    find_package(Qt5 5.15 REQUIRED COMPONENTS Core)
    set(QT_VERSION_MAJOR 5)
    set(QT_VERSION ${Qt5_VERSION})
endif()
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS ${KC_QT_COMPONENTS})
set(QtX Qt${QT_VERSION_MAJOR})
if(QT_VERSION_MAJOR GREATER_EQUAL 6)
    qt_standard_project_setup()
else()
    set(CMAKE_AUTOMOC ON)
endif()
set(CMAKE_AUTORCC ON)
message(STATUS "Building with Qt ${QT_VERSION}")

# Four-part version (Major.Minor.Build.Revision) used for the binaries and packages.
# scripts/build.ps1 passes -DKC_BUILD_VERSION=<version>.
set(KC_BUILD_VERSION "${PROJECT_VERSION}.0" CACHE STRING "Four-part version, e.g. 1.2.0.0")
if(NOT KC_BUILD_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)\\.([0-9]+)$")
    message(FATAL_ERROR "KC_BUILD_VERSION must look like 1.2.3.0, got '${KC_BUILD_VERSION}'")
endif()
set(KC_VER_MAJOR ${CMAKE_MATCH_1})
set(KC_VER_MINOR ${CMAKE_MATCH_2})
set(KC_VER_BUILD ${CMAKE_MATCH_3})
set(KC_VER_REVISION ${CMAKE_MATCH_4})
set(KC_SHORT_VERSION "${KC_VER_MAJOR}.${KC_VER_MINOR}.${KC_VER_BUILD}")
configure_file(res/version.h.in ${CMAKE_CURRENT_BINARY_DIR}/version.h @ONLY)
message(STATUS "Keyboard Changer version: ${KC_BUILD_VERSION}")

set(KC_SOURCES
    main.cpp
    src/core/types.h
    src/core/engine.h          src/core/engine.cpp
    src/core/presets.h         src/core/presets.cpp
    src/core/languages.h       src/core/languages.cpp       src/core/languages_gen.h
    src/core/textconvert.h     src/core/textconvert.cpp
    src/core/selectionconverter.h src/core/selectionconverter.cpp
    src/core/browserbridge.h   src/core/browserbridge.cpp
    src/platform/backend.h     src/platform/backend.cpp
    src/ui/icons.h             src/ui/icons.cpp
    src/ui/trayapp.h           src/ui/trayapp.cpp
    src/ui/settingswindow.h    src/ui/settingswindow.cpp
    res/resources.qrc
)

if(WIN32)
    list(APPEND KC_SOURCES
        src/platform/win/winbackend.h src/platform/win/winbackend.cpp
        src/platform/win/autostart_winrt.cpp
        res/app.rc)
elseif(APPLE)
    list(APPEND KC_SOURCES src/platform/mac/macbackend.h src/platform/mac/macbackend.mm res/app.icns)
    set_source_files_properties(src/platform/mac/macbackend.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    set_source_files_properties(res/app.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
else()
    find_package(X11 REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)
    list(APPEND KC_SOURCES
        src/platform/linux/linuxbackends.h
        src/platform/linux/linuxcommon.cpp
        src/platform/linux/x11backend.cpp
        src/platform/linux/dbusbackends.cpp
        src/platform/linux/wlrbackends.cpp)
endif()

# Translations (i18n/keyboard_changer_<code>.ts, generated by scripts/gen_i18n.py)
# compiled to .qm and embedded as :/i18n/keyboard_changer_<code>.qm
file(GLOB KC_TS_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/i18n/keyboard_changer_*.ts)
if(QT_VERSION_MAJOR GREATER_EQUAL 6)
    qt_add_executable(appkeyboard_changer WIN32 MACOSX_BUNDLE ${KC_SOURCES})
    qt_add_translations(appkeyboard_changer
        TS_FILES ${KC_TS_FILES}
        RESOURCE_PREFIX /i18n)
else()
    qt5_add_translation(KC_QM_FILES ${KC_TS_FILES})
    set(KC_QRC_BODY "")
    foreach(ts ${KC_TS_FILES})
        get_filename_component(n ${ts} NAME_WE)
        string(APPEND KC_QRC_BODY "<file>${n}.qm</file>")
    endforeach()
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/i18n.qrc
        "<RCC><qresource prefix=\"/i18n\">${KC_QRC_BODY}</qresource></RCC>\n")
    add_executable(appkeyboard_changer WIN32 MACOSX_BUNDLE ${KC_SOURCES}
        ${KC_QM_FILES} ${CMAKE_CURRENT_BINARY_DIR}/i18n.qrc)
endif()
target_include_directories(appkeyboard_changer PRIVATE src ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(appkeyboard_changer PRIVATE ${QtX}::Widgets ${QtX}::Network)

if(WIN32)
    set_target_properties(appkeyboard_changer PROPERTIES OUTPUT_NAME KeyboardChanger)
    target_compile_definitions(appkeyboard_changer PRIVATE
        WIN32_LEAN_AND_MEAN NOMINMAX
        # Windows 7 API level; newer functions are loaded at run time.
        _WIN32_WINNT=0x0601 WINVER=0x0601)
    target_link_libraries(appkeyboard_changer PRIVATE user32 dwmapi)
    # C++/WinRT StartupTask for autostart inside MSIX: MSVC + Windows SDK only.
    if(MSVC)
        target_link_libraries(appkeyboard_changer PRIVATE WindowsApp)
        target_compile_options(appkeyboard_changer PRIVATE /utf-8 /permissive- /bigobj)
        # Release: optimized binary + PDB (packed into .appxsym for Store crash reports)
        target_compile_options(appkeyboard_changer PRIVATE $<$<CONFIG:Release>:/Zi> $<$<CONFIG:Release>:/FS>)
        target_link_options(appkeyboard_changer PRIVATE
            $<$<CONFIG:Release>:/DEBUG:FULL> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
    endif()
    install(TARGETS appkeyboard_changer RUNTIME DESTINATION .)

elseif(APPLE)
    set_target_properties(appkeyboard_changer PROPERTIES
        OUTPUT_NAME "Keyboard Changer"
        MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in
        MACOSX_BUNDLE_GUI_IDENTIFIER io.github.keyboardchanger
        MACOSX_BUNDLE_BUNDLE_NAME "Keyboard Changer"
        MACOSX_BUNDLE_BUNDLE_VERSION ${KC_BUILD_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${KC_SHORT_VERSION}
        MACOSX_BUNDLE_ICON_FILE app.icns
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER io.github.keyboardchanger)
    target_link_libraries(appkeyboard_changer PRIVATE
        "-framework AppKit" "-framework Carbon" "-framework ServiceManagement")
    install(TARGETS appkeyboard_changer BUNDLE DESTINATION .)

else()
    set_target_properties(appkeyboard_changer PROPERTIES OUTPUT_NAME keyboard-changer)
    target_link_libraries(appkeyboard_changer PRIVATE ${QtX}::DBus X11::X11 X11::Xtst PkgConfig::XKBCOMMON)
    include(GNUInstallDirs)
    install(TARGETS appkeyboard_changer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES packaging/linux/io.github.KeyboardChanger.desktop
            DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES res/app.png RENAME io.github.KeyboardChanger.png
            DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
    # GNOME Shell extension: ES-module version for GNOME 45+, legacy one for GNOME 42-44
    # (Ubuntu 22.04, Debian 12). scripts/build-linux.sh picks it from `gnome-shell --version`.
    option(KC_GNOME_LEGACY_EXTENSION "Install the GNOME 42-44 extension instead of the 45+ one" OFF)
    if(KC_GNOME_LEGACY_EXTENSION)
        set(KC_GNOME_EXT_DIR gnome-extension-legacy)
    else()
        set(KC_GNOME_EXT_DIR gnome-extension)
    endif()
    install(DIRECTORY ${KC_GNOME_EXT_DIR}/keyboard-changer@keyboardchanger.github.io
            DESTINATION ${CMAKE_INSTALL_DATADIR}/gnome-shell/extensions)

    # cpack -G DEB / -G RPM / -G TGZ
    set(CPACK_PACKAGE_NAME keyboard-changer)
    set(CPACK_PACKAGE_VERSION ${KC_SHORT_VERSION})
    set(CPACK_PACKAGE_CONTACT "Keyboard Changer")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Switches the keyboard layout per application and website")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
    set(CPACK_RPM_PACKAGE_AUTOREQ ON)
    include(CPack)
endif()
